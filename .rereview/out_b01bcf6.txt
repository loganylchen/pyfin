    60	def _exons_from_candidate(c: TranscriptCandidate) -> Tuple[Tuple[int, int], ...]:
    61	    """Derive exon coordinates from a candidate's intron chain and boundaries."""
    62	    introns = c.intron_chain.introns
    63	    if not introns:
    64	        return ((c.start, c.end),)
    65	    exons = [(c.start, introns[0][0])]
    66	    for i in range(len(introns) - 1):
    67	        exons.append((introns[i][1], introns[i + 1][0]))
    68	    exons.append((introns[-1][1], c.end))
    69	    return tuple(exons)

    72	def quantify_transcripts(
    73	    R: np.ndarray,
    74	    hard_assignments: np.ndarray,
    75	    candidates: List[TranscriptCandidate],
    76	    read_ids: List[str],
    77	) -> List[QuantResult]:
    78	    """Compute probability-weighted transcript abundance (vectorized).
    79	
    80	    Args:
    81	        R: Soft assignment matrix of shape (n_reads, n_candidates).
    82	            R[i, j] = P(read_i belongs to candidate_j).
    83	        hard_assignments: Array of shape (n_reads,) with hard cluster indices.
    84	        candidates: List of TranscriptCandidate objects (column order matches R).
    85	        read_ids: List of read IDs (row order matches R).
    86	
    87	    Returns:
    88	        List of QuantResult, one per candidate.
    89	    """
    90	    n_reads, n_cands = R.shape
    91	
    92	    # Vectorized abundance: sum of soft assignments per candidate
    93	    abundance = R.sum(axis=0)  # shape (n_cands,)
    94	
    95	    # Vectorized hard assignment counts and confidence
    96	    read_ids_arr = np.asarray(read_ids)
    97	    results = []
    98	    for j, cand in enumerate(candidates):
    99	        assigned_mask = hard_assignments == j
   100	        num_assigned = int(assigned_mask.sum())
   101	        confidence = float(R[assigned_mask, j].mean()) if num_assigned > 0 else 0.0
   102	        assigned_ids = (
   103	            tuple(read_ids_arr[assigned_mask].tolist())
   104	            if num_assigned > 0
   105	            else ()
   106	        )
   107	
   108	        results.append(
   109	            QuantResult(
   110	                candidate_id=cand.candidate_id,
   111	                abundance=float(abundance[j]),
   112	                confidence=confidence,
   113	                num_assigned_reads=num_assigned,
   114	                source=cand.source,
   115	                chrom=cand.chrom,
   116	                strand=cand.strand,
   117	                start=cand.start,
   118	                end=cand.end,
   119	                exons=_exons_from_candidate(cand),
   120	                assigned_read_ids=assigned_ids,
   121	                breakpoint_left=cand.breakpoint_left,
   122	                breakpoint_right=cand.breakpoint_right,
   123	                fusion_junction=cand.fusion_junction,
   124	            )
   125	        )
   126	
   127	    return results

   130	def _qr_overlap(a: QuantResult, b: QuantResult) -> bool:
   131	    """1-D genomic interval overlap on (chrom, strand, [start, end))."""
   132	    if a.chrom != b.chrom or a.strand != b.strand:
   133	        return False
   134	    return a.start < b.end and b.start < a.end

   137	def isoform_fraction_drops(
   138	    results: Dict[str, QuantResult],
   139	    min_fraction: float,
   140	) -> set:
   141	    """Return candidate_ids to drop by the minimum-isoform-fraction heuristic.
   142	
   143	    For every NOVEL multi-exon transcript C, compute its locus-relative
   144	    abundance::
   145	
   146	        relabund(C) = C.abundance / max(abundance over ALL transcripts
   147	                       (novel, GTF and fusion) that OVERLAP C, C included)
   148	
   149	    and drop C when ``relabund(C) < min_fraction``. This is the standard
   150	    Cufflinks ``--min-isoform-fraction`` / StringTie ``-f`` minor-isoform
   151	    suppression rule: the low-fraction tail of a locus is enriched for
   152	    incompletely-spliced precursors (pre-mRNA), RT/template-switching
   153	    artifacts, and assembly noise rather than genuine isoforms.
   154	
   155	    The locus maximum spans ALL sources (not just novel): on dense annotated
   156	    loci the dominant isoform is usually a GTF-passthrough transcript, so
   157	    comparing a novel only against other novels lets wobble-shadow isoforms
   158	    (junction-shifted near-copies of a true transcript) escape — they only
   159	    compete with each other and keep a deceptively high novel-only fraction.
   160	    Measuring each novel against the strongest transcript of ANY source at its
   161	    locus collapses those shadows below the threshold.
   162	
   163	    GTF-passthrough (``source="gtf"``), fusion (``source="fusion"``), and
   164	    single-exon (mono) candidates are EXEMPT and never dropped — they only ever
   165	    contribute to the locus maximum, never to the drop set. ``min_fraction
   166	    <= 0`` disables the filter (returns an empty set).
   167	
   168	    SIRV WARNING: synthetic SIRV has no genuine low-abundance isoform tail, so
   169	    its F1-optimal threshold (~0.4) is 4-40x more aggressive than real tools
   170	    and would gut recall on a real transcriptome. Keep this conservative
   171	    (literature default 0.01) and re-tune on real data.
   172	    """
   173	    if min_fraction <= 0.0:
   174	        return set()

   176	    # strand can share a locus, so the pairwise overlap scan stays within a
   177	    # bucket instead of running all-vs-all across the whole genome. (Within a
   178	    # locus-dense single chrom/strand it is still O(n^2); a sweep-line could be
   179	    # added if a real run ever makes that the bottleneck.)

"""M2 junction NLL: per-event signal distance over internal-junction diff windows.

This is the production-shaped form of the validated "version B" wobble
discriminator (see ``experiments/_gt_zval_multi.py``; SIRV interval 10582-11643,
6 wobble pairs, pooled 86.2%). The signal engine is krill ``align_read_variants``
(in-memory eventalign, ``hmm_confidence=False``).

The three essential ingredients of the metric:
  1. NON-HMM eventalign (clean per-kmer ``standardized_level`` z for every event;
     ``model_kmer == reference_kmer`` is forced, so the only signal quantity is |z|).
  2. A SHARED genomic discrimination window restricted to *internal* intron
     boundaries (excludes 5'/3'/UTR structural differences), built from the
     exon symmetric difference between candidates of the same wobble class.
  3. per-event NLL = ``0.5·z² + log(model_stdv)``, **mean over the window**
     (÷n; the sum is read-length confounded).

M2 is an M1-structural-mask refinement: only (read, candidate) cells that M1
accepts (mappy aligns with no single indel > ``M1_MAX_INDEL_BP``) are scored;
M1-rejected cells get the sentinel ``MISSING`` distance. Cells whose candidate
class has no discrimination window (singletons / no internal diff) fall back to
the M1 distance, so M2 degrades to M1 where there is no wobble competition.
"""

@lru_cache(maxsize=16384)
def _build_exons_key(
    introns: Tuple[Tuple[int, int], ...], start: int, end: int
) -> Tuple[Tuple[int, int], ...]:
    """Cached exon geometry keyed on the hashable candidate signature.

    The exon layout is a pure function of (intron_chain, start, end), but the
    per-event NLL loop calls ``_tx2genome`` (hence ``_exon_tx_map`` ->
    ``_build_exons``) ~15M times against the SAME handful of candidates. Caching
    on the geometry tuple collapses those rebuilds to one per unique candidate.
    """
    exons: List[Tuple[int, int]] = []
    pos = start
    for s, e in sorted(introns):
        if s > pos:
            exons.append((pos, s))
        pos = max(pos, e)
    if end > pos:
        exons.append((pos, end))
    return tuple(exons)

- **Module-level mock/patch targets are not preserved.** M2 now resolves `em_with_coherence`, `krill_tiebreak`, and `quantify_transcripts` from `fin.pipeline.assignment` (`fin/pipeline/assignment.py:29`, `fin/pipeline/assignment.py:31`, `fin/pipeline/assignment.py:475`, `fin/pipeline/assignment.py:490`, `fin/pipeline/assignment.py:499`). Patching the previous targets in `fin.pipeline.runner` no longer intercepts M2 execution. The commit updates two in-repo targets, but silently also moves the `krill_tiebreak` seam. Because runner still exports these names for other modes, old patches do not fail loudly—they simply stop controlling M2 and may execute real EM/quantification/signal code.

The moved arithmetic, thresholds, kept-read logic, drop ordering, containment fold, and lazy observed-junction construction otherwise match the parent. Syntax compilation passed; focused pytest collection was unavailable in the provided environment because its pytest interpreter lacks NumPy.
