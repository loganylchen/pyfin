   214	def cluster_read_chains(
   215	    read_chains: List[Tuple[Dict, IntronChain]],
   216	    *,
   217	    wobble_bp: int = 6,
   218	    cassette_max_exon_bp: int = 70,
   219	    fold_monoexon_contained: bool = False,
   220	    fold_span_guard: bool = False,
   221	) -> List[ChainCluster]:
   222	    """Build generation-side clusters from per-read chains (NO snap, 3' ignored).
   223	
   224	    Args:
   225	        read_chains: per-read ``(read_dict, IntronChain)`` (read_dict needs
   226	            "query_name"; CIGAR-derived chains, used as-is).
   227	        wobble_bp: per-junction tolerance for wobble / cassette / containment joins.
   228	        cassette_max_exon_bp: max skipped-exon length for a cassette join (0 off).
   229	        fold_monoexon_contained: when True, a single-exon (chain-less) read whose
   230	            aligned span lies wholly inside one exon of a multi-exon candidate is folded
   231	            into that candidate (its read joins the candidate's support) instead of
   232	            emitting a standalone mono candidate. This absorbs 5'/3' degradation
   233	            fragments of a spliced transcript. Reads inside an INTRON (a different gene)
   234	            or not contained in any multi-exon candidate stay as mono candidates. Off by
   235	            default (byte-identical legacy behaviour).
   236	        fold_span_guard: when True, a read only folds into an exact-sub-chain container
   237	            if its aligned span does NOT run exonically across one of the container's
   238	            EXTRA introns (introns present in the container but absent from the read's
   239	            chain). A read that spans such an intron contradicts it (retained-intron or
   240	            alternative isoform) and is kept as its own candidate instead of being
   241	            absorbed. Off by default (byte-identical legacy behaviour).
   242	
   243	    Returns:
   244	        List of :class:`ChainCluster`. EVERY exact contiguous sub-chain is folded
   245	        (unconditionally, no read-support guard) into its longest exact container --
   246	        generation does pure structural collapse and does NOT decide whether a folded
   247	        sub-chain is a real short isoform; that judgement is a SEPARATE downstream
   248	        recovery step (e.g. read-end pileup), not done here. The survivors are unioned
   249	        into clusters by wobble/cassette/containment; each unique mono chain is its own
   250	        singleton cluster.
   251	    """
   252	    # 1. group reads by EXACT chain (no 3', no snap).
   253	    by_chain: Dict[Chain, Set[str]] = {}
   254	    for rd, chain in read_chains:
   255	        rid = rd.get("query_name")
   256	        if rid is None:
   257	            continue
   258	        by_chain.setdefault(tuple(chain.introns), set()).add(rid)
   259	
   260	    # Read spans (0-based [start, end)) -- needed by the span-guarded fold and the
   261	    # mono-exon fold. Built once, only when a span-aware step is enabled.
   262	    read_span: Dict[str, Tuple[int, int]] = {}
   263	    if fold_span_guard or fold_monoexon_contained:
   264	        for rd, _chain in read_chains:
   265	            rid = rd.get("query_name")
   266	            rs = rd.get("reference_start")
   267	            re_ = rd.get("reference_end")
   268	            if rid is not None and rs is not None and re_ is not None:
   269	                read_span[rid] = (rs, re_)
   270	
   271	    # 2. collapse every exact contiguous sub-chain into its longest exact container.
   272	    #    Default: UNCONDITIONAL (no read-support guard). With fold_span_guard, a read
   273	    #    folds ONLY if its aligned span does not run exonically across one of the
   274	    #    container's EXTRA introns (retained-intron / alternative isoform reads are kept
   275	    #    as their own candidate instead of being absorbed). Whether a *consistent*
   276	    #    folded sub-chain is a genuine short isoform is still a SEPARATE downstream
   277	    #    recovery step (read-end pileup), never guessed here from read counts.
   278	    chains_sorted = sorted(by_chain, key=lambda c: (-len(c), -len(by_chain[c]), c))
   279	    kept: List[Chain] = []
   280	    reads: Dict[Chain, Set[str]] = {}
   281	    folded_of: Dict[Chain, List[Tuple[Chain, Set[str]]]] = {}  # container -> shadows
   282	    for c in chains_sorted:
   283	        container = next((k for k in kept if _exact_subchain(c, k)), None)
   284	        if container is None:
   285	            kept.append(c)
   286	            reads[c] = set(by_chain[c])
   287	            continue
   288	        if not fold_span_guard:
   289	            reads[container] |= by_chain[c]
   290	            folded_of.setdefault(container, []).append((c, set(by_chain[c])))
   291	            continue
   292	        # span guard: split c's reads into those consistent with the container (fold)
   293	        # and those exonic across one of the container's extra introns (keep as c).
   294	        off = _subchain_offset(c, container)
   295	        extra = container[:off] + container[off + len(c):]
   296	        fold_r: Set[str] = set()
   297	        keep_r: Set[str] = set()
   298	        for rid in by_chain[c]:
   299	            sp = read_span.get(rid)
   300	            if sp is not None and any(sp[0] <= d and sp[1] >= a for d, a in extra):
   301	                keep_r.add(rid)            # exonic across an extra intron -> distinct
   302	            else:
   303	                fold_r.add(rid)
   304	        if fold_r:
   305	            reads[container] |= fold_r
   306	            folded_of.setdefault(container, []).append((c, fold_r))
   307	        if keep_r:

   310	
   311	    # 3. union-find the survivors by wobble / cassette / containment (multi-exon only).

   314	
   315	    # 3b. (optional) fold single-exon reads that are wholly inside a multi-exon
   316	    #     candidate's exon into that candidate -- 5'/3' degradation fragments of a
   317	    #     spliced transcript, otherwise emitted as standalone mono FPs. Per-read (the
   318	    #     mono chain pools every chain-less read), deterministic container choice.

   320	        multi_span: Dict[Chain, Tuple[int, int]] = {}
   321	        for c in multi:
   322	            pts = [read_span[r] for r in reads[c] if r in read_span]
   323	            if pts:
   324	                multi_span[c] = (min(p[0] for p in pts), max(p[1] for p in pts))
   325	        containers = [c for c in multi if c in multi_span]
   326	        for mc in mono:
   327	            remaining: Set[str] = set()
   328	            for rid in reads[mc]:
   329	                sp = read_span.get(rid)
   330	                if sp is None:
   331	                    remaining.add(rid)
   332	                    continue
   333	                s, e = sp
   334	                hits = [c for c in containers
   335	                        if _monoexon_in_exon(s, e, c, *multi_span[c])]
   336	                if not hits:
   337	                    remaining.add(rid)
   338	                    continue
   339	                best = max(hits, key=lambda c: (len(reads[c]), len(c), c))
   340	                reads[best].add(rid)
   341	                folded_of.setdefault(best, []).append(((), {rid}))
   342	            reads[mc] = remaining
   343	        mono = [c for c in mono if reads[c]]
   344	
   345	    parent = {c: c for c in multi}
   346	
   347	    def find(x: Chain) -> Chain:
   348	        while parent[x] != x:
   349	            parent[x] = parent[parent[x]]
   350	            x = parent[x]
   351	        return x
   352	
   353	    for i in range(len(multi)):
   354	        for j in range(i + 1, len(multi)):
   355	            a, b = multi[i], multi[j]
   356	            if find(a) != find(b) and _related(a, b, wobble_bp, cassette_max_exon_bp):
   357	                parent[find(a)] = find(b)
   358	
   359	    comps: Dict[Chain, List[Chain]] = {}
   360	    for c in multi:
   361	        comps.setdefault(find(c), []).append(c)
   362	
   363	    def _mk(c: Chain) -> GenCandidate:
   364	        """Member candidate with its folded exact-sub-chain shadows attached."""
   365	        return GenCandidate(
   366	            IntronChain(introns=c), set(reads[c]),
   367	            folded=[GenCandidate(IntronChain(introns=fc), set(fr))
   368	                    for fc, fr in folded_of.get(c, [])])
   369	
   370	    clusters: List[ChainCluster] = []
   371	    for comp in comps.values():
   372	        members = [_mk(c) for c in comp]
   373	        union: Set[str] = set()
   374	        for c in comp:
   375	            union |= reads[c]
   376	        clusters.append(ChainCluster(members=members, read_ids=union))
   377	
   378	    for c in mono:  # mono singletons
   379	        clusters.append(ChainCluster(members=[_mk(c)], read_ids=set(reads[c])))
   380	
   381	    return clusters

   394	def _attach_gtf_variants(
   395	    families: List[ChainFamily],
   396	    gtf_variants: List[Tuple[str, Chain]],

   398	    cassette_max_exon_bp: int,

   401	
   402	    NON-BRIDGING: a GTF is added to at most ONE family and never merges two read families. A GTF
   403	    related to no read family becomes its own GTF-only family (empty read pool). Mono (empty-chain)

   410	    gtf_only: List[ChainFamily] = []

   412	        if not gchain:                     # mono / empty-chain GTF -> deferred to the mono finalizer
   413	            continue

   415	        for fam in read_families:

   417	                    if _related(gchain, v, wobble_bp, cassette_max_exon_bp))
   418	            if n == 0:
   419	                continue

   427	                gtf_members=[(gid, gchain)]))
   428	    families.extend(gtf_only)

   164	def _cassette(a: Chain, b: Chain, bp: int, max_exon_bp: int) -> bool:
   165	    """``a`` (K introns) is a cassette-skip sibling of ``b`` (K-1 introns): at one
   166	    position ``i`` the two introns ``a[i], a[i+1]`` replace ``b[i]``'s span (outer
   167	    donor/acceptor within bp), the skipped exon between them is < ``max_exon_bp``, and
   168	    every other intron matches within bp."""
   169	    if max_exon_bp <= 0 or len(a) != len(b) + 1:
   170	        return False
   171	    for i in range(len(a) - 1):
   172	        if any(abs(a[k][0] - b[k][0]) > bp or abs(a[k][1] - b[k][1]) > bp
   173	               for k in range(i)):
   174	            continue
   175	        if abs(a[i][0] - b[i][0]) > bp or abs(a[i + 1][1] - b[i][1]) > bp:
   176	            continue
   177	        rest = len(b) - i - 1
   178	        if any(abs(a[i + 2 + k][0] - b[i + 1 + k][0]) > bp or
   179	               abs(a[i + 2 + k][1] - b[i + 1 + k][1]) > bp for k in range(rest)):
   180	            continue
   181	        cas = a[i + 1][0] - a[i][1]

   183	            return True
