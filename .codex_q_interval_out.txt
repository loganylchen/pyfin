     1	"""Interval-level process parallelism for the assembly pipeline.
     2	
     3	The per-interval work (:meth:`PipelineRunner.process_interval`) is wholly
     4	self-contained and the cross-interval aggregation
     5	(:func:`aggregate_across_intervals`) is commutative, so intervals can be run in
     6	independent worker processes and merged in any order.
     7	
     8	Concurrency model (see ``fin/pipeline/runner.py`` ``run()``):
     9	  * ``spawn`` start method -- the parent has already imported numpy, so a forked
    10	    worker would inherit the parent's BLAS thread pool and the per-worker thread
    11	    caps set in ``cli.main()`` would not take effect; spawn re-imports fresh.
    12	    CUDA / pysam / Pod5 handles are also not fork-safe.
    13	  * Static GPU/CPU partition -- exactly ``gpu_workers`` of the ``threads`` workers
    14	    build a GPU-backed runner (``config.use_gpu = True``); the rest are CPU-only.
    15	    This bounds live CUDA contexts to ``gpu_workers`` (a semaphore over calls would
    16	    not -- contexts would leak to ``threads``). ``make_krill_aligner`` auto-falls
    17	    back to CPU per worker on GPU-init/OOM.
    18	  * Each worker builds ONE runner in the pool initializer and reuses it across every
    19	    interval it is handed; interval indices (not objects) cross the boundary, and
    20	    each worker rebuilds the deterministic interval list itself.
    21	"""

    23

    89	def run_parallel(
    90	    config: PipelineConfig,
    91	    intervals: List,
    92	    threads: int,
    93	    gpu_workers: int,
    94	    log_level: str,

    97	
    98	    Returns the list of non-empty per-interval result lists (order arbitrary;
    99	    downstream aggregation is commutative).
   100	    """
   101	    n = len(intervals)
   102	    ctx = mp.get_context("spawn")
   103	    counter = ctx.Value("i", 0)
   104	    all_results: List[List[QuantResult]] = []
   105	
   106	    ex = ProcessPoolExecutor(
   107	        max_workers=threads,
   108	        mp_context=ctx,

   114	        done = 0

   117	            done += 1
   118	            i = futures[fut]

   120	                "Completed interval %d/%d: %s", done, n, intervals[i].region_string
   121	            )
   122	            if quant:

   124	    except BaseException:

   126	        # futures and return without blocking on the remaining workload (a plain
   127	        # finally: shutdown(wait=True) would stall the failure behind every queued
   128	        # interval). Already-running tasks cannot be force-killed via the executor;
   129	        # they finish in their own process but the parent no longer waits on them.
   130	        ex.shutdown(wait=False, cancel_futures=True)
   131	        raise
   132	    else:
   133	        ex.shutdown(wait=True)
   134	    return all_results
